# Домашнее задание к работе 10

## Условие задачи
Составьте программу, которая для заданной пользователем фигуры (равнобедренный треугольник) предлагает меню выбора одной из операций:
1) рассчитать площадь;
2) вывести определение фигуры;
3) нарисовать фигуру.

## 1. Алгоритм и блок-схема

### Алгоритм
1. **Начало**.
2. Выводим на экран меню с тремя пунктами и просим пользователя ввести номер действия.
3. Считываем выбор `choice`.
4. В зависимости от выбора:
   - Если `choice == 1` (площадь): вызываем функции `get_h()` и `get_a()` для ввода высоты и основания, затем считаем площадь через `area(a, h)` и выводим результат.
   - Если `choice == 2` (определение): выводим на экран текстовое определение равнобедренного треугольника.
   - Если `choice == 3` (рисование): просим ввести символ `sim`, запрашиваем высоту `h` и вызываем функцию `draw(h, sim)` для отрисовки контура.
   - Иначе: выводим сообщение о неверном выборе.
5. **Конец**.

**Часть 2: Функции**
- `float get_h()`: запрашивает у пользователя высоту и возвращает её.
- `float get_a()`: запрашивает основание и возвращает его.
- `float area(float a, float h)`: возвращает площадь по формуле S = 0.5 * a * h.
- `void draw(int h, char sim)`: рисует контур равнобедренного треугольника заданной высоты с помощью введенного символа.
- `int compare(float a1, float h1, float a2, float h2)`: сравнивает площади двух треугольников и возвращает 1 (если первый больше), -1 (если второй больше) или 0 (если равны). *(Добавлено для соответствия заданию)*.

## 2. Реализация программы

```c
#include <stdio.h>
#include <locale.h>

float area(float a, float h);
char get_sim(int i, int j, int h, char sim); 
int compare(float a1, float h1, float a2, float h2);

int main() {
    int choice;
    char sim;
    float h, a;
    setlocale(LC_ALL, ".UTF8");

    printf("--- дз ---\n");
    printf("1. площадь\n2. определение\n3. нарисовать\n");
    printf("выберите число от 1 до 3: ");
    scanf("%d", &choice);

    if (choice == 1) {
        printf("введите высоту h: ");
        scanf("%f", &h);
        printf("введите основание a: ");
        scanf("%f", &a);
        
        if (h <= 0 || a <= 0) {
            printf("Ошибка: размеры должны быть положительными\n");
        } else {
            printf("Площадь: %.2f\n", area(a, h));
        }
    } 
    else if (choice == 2) {
        printf("Равнобедренный треугольник — это треугольник, у которого две стороны равны.\n");
        printf("Углы при основании у него также равны.\n");
    } 
    else if (choice == 3) {
        printf("введите высоту h: ");
        scanf("%f", &h);
        printf("введите символ: ");
        scanf(" %c", &sim);
        
        for (int i = 1; i <= (int)h; i++) {
            for (int j = 0; j < 2 * i - 1; j++) {
                printf("%c", get_sim(i, j, (int)h, sim));
            }
            printf("\n");
        }
    } 
    else {
        printf("неверный выбор.\n");
    }

    return 0;
}

float area(float a, float h) {
    return 0.5f * a * h;
}

char get_sim(int i, int j, int h, char sim) {
    if (j == 0 || j == 2 * i - 2 || i == h) {
        return sim;
    } else {
        return ' ';
    }
}

int compare(float a1, float h1, float a2, float h2) {
    float s1 = area(a1, h1);
    float s2 = area(a2, h2);
    if (s1 > s2) return 1;
    if (s1 < s2) return -1;
    return 0;
}
```
